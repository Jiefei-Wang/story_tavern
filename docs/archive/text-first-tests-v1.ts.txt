function adapter(options: {
    channels?: Partial<Delivery>;
    noPerception?: boolean;
    badRoute?: boolean;
    fail?: string;
    inspect?: (o: RunAgentOptions) => void;
} = {}) {
    const calls: RunAgentOptions[] = [];
    const run = async (o: RunAgentOptions) => {
        calls.push(structuredClone({ ...o, signal: undefined }));
        options.inspect?.(o);
        const { task, material } = o.context;
        if (options.fail === o.agentId)
            throw new Error('controlled model failure');
        let data: any;
        if (o.agentId === 'text_narrator')
            data = '来客放下陌生物件。艾琳暂时没有开口。';
        else if (o.agentId === 'text_character')
            data = o.conversation?.at(-1)?.content.startsWith('第一轮') ? '我不知道这是什么，先观察。' : o.conversation?.at(-1)?.content.startsWith('第二轮') ? '保持沉默。' : '没有新的动作。';
        else if (o.agentId === 'text_editor') {
            const conversation = o.conversation || [];
            const path = material.paths.find((p: string) => p.startsWith('scenes/')) || material.paths[0];
            const isMemory = path.endsWith('/memory.md');
            const desired = isMemory ? '\n本轮个人经历：' + JSON.stringify(material.material) : task.startsWith('变暗') ? '\n光照：黑暗。' : task.startsWith('记录玩家') ? '\n已经发生的玩家言行：' + material.material.player : '';
            if (!desired)
                data = { op: 'done' };
            else if (!conversation.length)
                data = { op: 'read', path };
            else if (conversation.length === 2) {
                const doc = JSON.parse(conversation.at(-1)!.content.replace('工具结果：', ''));
                data = { op: 'replace', path, expected_revision: doc.revision, old_text: doc.text, new_text: doc.text + desired };
            }
            else
                data = { op: 'done' };
        }
        else if (task.startsWith('为玩家'))
            data = { background: '清晨的酒馆门外，艾琳站在来客旁边。', actions: material.input, speech: '', correction: null, clarification: null, new_people: [] };
        else if (task.startsWith('输入整理'))
            data = options.badRoute ? { before: ['forged'], after: [], interpretation: '伪造' } : { before: material.instructions.filter((i: any) => i.kind === 'admin' && (i.text.includes('之前') || i.defaultPosition === 'before')).map((i: any) => i.id), after: material.instructions.filter((i: any) => i.kind === 'admin' && !i.text.includes('之前') && i.defaultPosition === 'after').map((i: any) => i.id), interpretation: '完整言行单位', scope: Object.fromEntries(material.instructions.filter((i: any) => i.kind === 'admin').map((i: any) => [i.id, ['scenes/current.md']])) };
        else if (task.startsWith('仅描述'))
            data = { text: '环境变暗。' };
        else if (task.startsWith('感知')) {
            const player = material.event.startsWith('我');
            const dark = material.scene.includes('光照：黑暗');
            data = { deliveries: !options.noPerception && player ? [{ character_id: 'erin', seen: dark ? null : '来客放下掌心大小的光滑物件。', heard: null, felt: null, environment: '门外', common: '只认识古代日常器物', ...options.channels }] : [] };
        }
        else if (task.startsWith('局部先后'))
            data = { next: material.pending[0]?.id || null, performance: material.pending[0] ? material.pending[0].speech + ' ' + material.pending[0].action : null };
        else if (task.startsWith('只找'))
            data = { continue: [] };
        else
            throw new Error('Unhandled test task ' + task);
        return { success: true, data: typeof data === 'string' ? data : JSON.stringify(data), spanId: 'test' };
    };
    return { run, calls };
}
test('authority: whole-line blocks preserve a single player unit, offsets, defaults and literal text', () => {
    const input = '[admin]\n前置\n[/admin]\n我进屋\n[admin]\n中间\n[/admin]\n说“你好”。\n[narrator]\n简短\n[/narrator]';
    const parsed = parseTextInput(input);
    assert.equal(parsed.player, '我进屋\n说“你好”。');
    assert.equal(parsed.instructions.length, 3);
    assert.deepEqual(parsed.instructions.map(i => i.defaultPosition), ['before', 'after', 'after']);
    for (const i of parsed.instructions)
        assert(input.slice(i.start, i.end).includes(i.text));
    for (const literal of ['```text\n[admin]\n秘密\n[/admin]\n```', '> [admin]\n> 秘密\n> [/admin]', '我说：“\n[admin]\n秘密\n[/admin]\n”', '\\[admin]\n只是示例\n\\[/admin]'])
        assert.equal(parseTextInput(literal).instructions.length, 0);
    assert.equal(parseTextInput('admin: 修改光照').instructions[0].text, '修改光照');
    assert.equal(parseTextInput('管理员: 修改光照').player, '');
    assert.throws(() => parseTextInput('[admin]\n未闭合'), /未闭合/);
});
